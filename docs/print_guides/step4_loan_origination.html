@router.post("", response_model=LoanDetailResponse, status_code=status.HTTP_201_CREATED)
async def create_one_loan(payload: LoanCreate, db: DbSession, current_user: CurrentUser) -> LoanDetailResponse:
    # 1. Idempotency Check: Prevent duplicate loan origination
    if payload.request_id is not None:
        existing = await loan_service.get_loan_by_request_id(db, payload.request_id)
        if existing is not None: return LoanDetailResponse.model_validate(existing)

    # 2. Verify Borrower entity exists
    borrower = await borrower_service.get_borrower(db, payload.borrower_id)
    if borrower is None: raise HTTPException(status_code=404, detail="Borrower not found")

    # 3. Create Loan + compute reducing balance amortization schedule
    created_loan = await loan_service.create_loan(db, payload, current_user)
    await db.commit()
    return LoanDetailResponse.model_validate(created_loan)
Amortization Engine — loan_calculator.py
from decimal import Decimal, ROUND_HALF_UP
CENT = Decimal("0.01")

def calculate_period_interest(principal: Decimal, rate: Decimal) -> Decimal:
    return (principal * rate).quantize(CENT, rounding=ROUND_HALF_UP)

def calculate_reducing_balance_schedule(principal: Decimal, rate: Decimal, count: int) -> list[Installment]:
    schedule = []
    rem = principal
    pmt = (principal * (rate + (rate / (((1 + rate)**count) - 1)))).quantize(CENT)
    for i in range(1, count + 1):
        interest = calculate_period_interest(rem, rate)
        prin = pmt - interest
        rem -= prin
        schedule.append(Installment(number=i, payment_amount=pmt, interest_amount=interest, principal_amount=prin, remaining_principal=rem))
    return schedule
Financial Rule: Always use Python `Decimal` with `ROUND_HALF_UP` cent quantization to eliminate IEEE floating-point errors.

Step 4 Debugging & Troubleshooting Guide

Troubleshooting Checklist
Issue / Error Symptom Root Cause Backend / Database Fix
Penny Off Error on Last Installment Accumulated cent rounding mismatch over multiple periods Adjust final installment payment to exactly match remaining principal balance.
HTTP 409 "Request ID used for different terms" Network retry sent modified loan parameters with same request_id Enforce strict parameter equality check when handling duplicate idempotency keys.
Check Constraint Violation `ck_loans_status` Attempting to set status string outside allowed enum list Ensure status is one of 'Draft', 'Active', 'Paid', 'Overdue'.
PostgreSQL Decimal Truncation Warning Passing float value instead of Python Decimal string Parse all request values using Decimal(str(val)) before database writes.
