Layer 2: FastAPI Login Endpoint & Auth Router

FastAPI Auth Router — backend/app/routers/auth.py
Login Endpoint — auth.py (Lines 56-77)
@router.post("/token", response_model=TokenResponse)
async def login(payload: LoginRequest, db: DbSession) -> TokenResponse:
    # Step 1 — Authenticate username & verify bcrypt password hash against PostgreSQL
    user = await authenticate_user(db, payload.username, payload.password)

    # Step 2 — Reject invalid credentials with generic 401 Unauthorized (prevents user enumeration)
    if user is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid username or password",
        )

    # Step 3 — Issue short-lived access_token (15m) & long-lived refresh_token (7d)
    return TokenResponse(
        access_token=create_token(user, "access"),
        refresh_token=create_token(user, "refresh"),
    )

Layer 3: Backend Authentication & bcrypt Password Hashing Service

Auth Logic & Passlib bcrypt Service — backend/app/services/auth_service.py
Authentication Logic — auth_service.py (Lines 28-56)
from passlib.context import CryptContext
password_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

def verify_password(password: str, hashed_password: str) -> bool:
    # 1. Enforce bcrypt 72-byte limit to prevent DoS attacks
    if len(password.encode("utf-8")) > 72:
        return False
    # 2. Compare plain password with stored bcrypt hash
    return password_context.verify(password, hashed_password)

async def authenticate_user(db: AsyncSession, username: str, password: str) -> User | None:
    # 3. Query user by username and verify bcrypt hash
    result = await db.execute(select(User).where(User.username == username))
    user = result.scalar_one_or_none()
    if user is None or not verify_password(password, user.hashed_password):
        return None
    return user

Layer 5: Mobile App HTTP Login & Storage Integration

Flutter Client Authentication — lib/features/auth/data/auth_repository.dart
AuthRepository Login Method — auth_repository.dart
Future<void> login(String username, String password) async {
  // 1. Send POST request to /api/v1/auth/token via Dio
  final response = await _dio.post<Map<String, dynamic>>(
    ApiEndpoints.token,
    data: {'username': username, 'password': password},
  );
  
  // 2. Extract access and refresh JWT strings from response body
  final accessToken = response.data?['access_token'] as String?;
  final refreshToken = response.data?['refresh_token'] as String?;
  
  // 3. Save JWT tokens into encrypted FlutterSecureStorage
  await Future.wait([
    _secureStorage.write(key: 'access_token', value: accessToken),
    _secureStorage.write(key: 'refresh_token', value: refreshToken),
  ]);
}

Step 1 Debugging & Security Checklist

Troubleshooting & Security Checklist
Issue / Error Symptom Root Cause Backend / Database Fix
HTTP 401 "Invalid username or password" Username not in DB or bcrypt password mismatch Run python -m app.bootstrap to seed initial default users.
HTTP 422 "Unprocessable Entity" Missing username or password key in JSON request body Check Postman JSON body payload matches LoginRequest schema keys.
ValueError "Password exceeds 72 UTF-8 bytes" Input password string exceeds bcrypt length limit Sanitize or truncate input string before passing to bcrypt.
JWT Signature Error on Subsequent Endpoints Mismatched SECRET_KEY between token issue and verify Ensure SECRET_KEY in backend/.env remains constant.
