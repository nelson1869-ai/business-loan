Layer 2: FastAPI Borrower Router & Pagination Headers

FastAPI Router — backend/app/routers/borrowers.py
Paginated Listing & Creation Endpoints — borrowers.py
@router.get("", response_model=list[BorrowerResponse])
async def list_all_borrowers(
    db: DbSession, current_user: CurrentUser, response: Response,
    borrower_status: Annotated[str | None, Query(alias="status")] = None,
    offset: Annotated[int, Query(ge=0)] = 0,
    limit: Annotated[int, Query(ge=1, le=100)] = 50,
) -> list[BorrowerResponse]:
    # 1. Query paginated list & total row count from database
    borrowers, total = await borrower_service.list_borrowers(db, borrower_status, offset, limit)
    # 2. Expose total count in HTTP Header X-Total-Count for mobile list pagination
    response.headers["X-Total-Count"] = str(total)
    return [BorrowerResponse.model_validate(item) for item in borrowers]

@router.post("", response_model=BorrowerResponse, status_code=status.HTTP_201_CREATED)
async def register_borrower(payload: BorrowerCreate, db: DbSession, current_user: CurrentUser) -> BorrowerResponse:
    try:
        # 3. Create borrower record & stage redacted audit log in same SQL transaction
        borrower = await borrower_service.create_borrower(db, payload, current_user)
        await db.commit()
    except IntegrityError as error:
        await db.rollback()
        raise HTTPException(status_code=status.HTTP_409_CONFLICT, detail="National ID already exists")
    return BorrowerResponse.model_validate(borrower)

PII Redaction & Immutable Audit Trail — borrower_service.py
def _audit_state(borrower: Borrower) -> str:
    # 1. PII Protection: Redact names, phone numbers, and national IDs before audit writing
    return json.dumps({
        "id": borrower.id,
        "status": borrower.status,
        "firstName": "[REDACTED]",
        "lastName": "[REDACTED]",
        "nationalId": "[REDACTED]",
        "phone": "[REDACTED]",
    })

async def create_borrower(db: AsyncSession, payload: BorrowerCreate, user: User) -> Borrower:
    # 2. Construct new Borrower entity
    borrower = Borrower(id=str(uuid4()), **payload.model_dump())
    db.add(borrower)
    
    # 3. Write immutable audit log record in same SQL transaction
    db.add(AuditLog(
        id=str(uuid4()), user_id=user.id, action="CREATE_BORROWER",
        entity_name="borrowers", entity_id=borrower.id,
        old_state_json=None, new_state_json=_audit_state(borrower)
    ))
    return borrower
Security Constraint: PII (Name, Phone, National ID) is NEVER saved in cleartext inside system audit logs. All mutation logs redact PII values to comply with privacy laws.

Layer 4: PostgreSQL Schema, Indexes & Audit Tables

SQL Persistence — backend/app/models/borrower.py & audit_log.py
Borrower Table Schema — models/borrower.py
class Borrower(Base):
    __tablename__ = "borrowers"

    id = mapped_column(String(36), primary_key=True)
    first_name = mapped_column(String(100), nullable=False)
    last_name = mapped_column(String(100), nullable=False)
    national_id = mapped_column(String(100), nullable=False, unique=True, index=True)
    phone = mapped_column(String(32), nullable=False)
    status = mapped_column(String(20), nullable=False, index=True)
    created_at = mapped_column(DateTime(timezone=True), server_default=func.now())
Immutable Audit Log Schema — models/audit_log.py
class AuditLog(Base):
    __tablename__ = "audit_logs"

    id = mapped_column(String(36), primary_key=True)
    user_id = mapped_column(UUID(as_uuid=True), ForeignKey("users.id"))
    action = mapped_column(String(50), nullable=False)
    entity_name = mapped_column(String(50), nullable=False)
    entity_id = mapped_column(String(36), nullable=False)
    new_state_json = mapped_column(Text, nullable=True)
    created_at = mapped_column(DateTime(timezone=True), server_default=func.now())
SQL Performance: B-Tree indexes on national_id and status enable sub-millisecond lookup speeds on 100,000+ borrower records!

Masked PII Text Widget — pii_masked_text.dart
/// Displays masked PII until user toggles reveal
class PiiMaskedText extends StatefulWidget {
  final String value;
  const PiiMaskedText({super.key, required this.value});

  @override
  State<PiiMaskedText> createState() => _PiiMaskedTextState();
}

class _PiiMaskedTextState extends State<PiiMaskedText> {
  bool _isVisible = false;

  @override
  Widget build(BuildContext context) {
    final display = _isVisible ? widget.value : '••••••••${widget.value.characters.takeLast(4)}';
    return InkWell(
      onTap: () => setState(() => _isVisible = !_isVisible),
      child: Text(display, style: const TextStyle(fontFamily: 'monospace')),
    );
  }
}

Step 3 Debugging & Troubleshooting Guide

Troubleshooting Checklist
Issue / Error Symptom Root Cause Backend / Database Fix
HTTP 409 "National ID already exists" Unique index constraint violation on national_id column Catch IntegrityError in FastAPI router and return HTTP 409 Conflict.
Mobile List Missing Total Page Count FastAPI router not returning X-Total-Count header Ensure response.headers["X-Total-Count"] = str(total) is set.
PII Data Leaked in Audit Logs Unsanitized model passed to AuditLog.new_state_json Always process entities through _audit_state() before staging log.
PostgreSQL Slow Query on Status Search Missing B-Tree index on status column Add index=True to status = mapped_column(...) in borrower model.
